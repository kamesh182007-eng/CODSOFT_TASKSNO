4. Segment customers (age, location and other traits)

def profile(by):
    """Size, share and high-income rate for any grouping column."""
    g = df.groupby(by, observed=True).agg(
        customers=("age", "size"),
        high_income_rate=("high_income", "mean"),
        avg_age=("age", "mean"),
        avg_hours=("hours.per.week", "mean"),
        avg_capital_gain=("capital.gain", "mean"),
    )
    g["share_of_customers"] = g["customers"] / g["customers"].sum()
    g["share_of_all_high_income"] = (df.groupby(by, observed=True)["high_income"].sum()
                                     / df["high_income"].sum())
    return g.round(3)

5. Buying-pattern segmentation with K-Means

features = pd.DataFrame({
    "age": df["age"],
    "education.num": df["education.num"],
    "hours.per.week": df["hours.per.week"],
    "log_capital_gain": np.log1p(df["capital.gain"]),
})
X = StandardScaler().fit_transform(features)

# Elbow method + silhouette score to choose the number of clusters
ks = range(2, 9)
inertia, sil = [], []
sample = np.random.RandomState(42).choice(len(X), 5000, replace=False)
for k in ks:
    km = KMeans(n_clusters=k, n_init=10, random_state=42).fit(X)
    inertia.append(km.inertia_)
    sil.append(silhouette_score(X[sample], km.labels_[sample]))

fig, ax = plt.subplots(1, 2, figsize=(11, 4))
ax[0].plot(list(ks), inertia, marker="o")
ax[0].set(title="Elbow Method", xlabel="Number of clusters (k)", ylabel="Inertia")
ax[1].plot(list(ks), sil, marker="o", color="tab:orange")
ax[1].set(title="Silhouette Score", xlabel="Number of clusters (k)", ylabel="Score")
plt.tight_layout()
plt.show()
K = 4                                              # chosen from the elbow chart
df["cluster"] = KMeans(n_clusters=K, n_init=10, random_state=42).fit_predict(X)

cl = df.groupby("cluster").agg(
    customers=("age", "size"),
    avg_age=("age", "mean"),
    avg_education=("education.num", "mean"),
    avg_hours=("hours.per.week", "mean"),
    avg_capital_gain=("capital.gain", "mean"),
    high_income_rate=("high_income", "mean"),
).round(2)
cl["value_rank"] = cl["high_income_rate"].rank(ascending=False).astype(int)
cl["share_of_customers"] = (cl["customers"] / cl["customers"].sum()).round(3)
cl = cl.sort_values("value_rank")

labels = {c: f"Cluster {c} (rank {cl.loc[c, 'value_rank']})" for c in cl.index}
df["cluster_label"] = df["cluster"].map(labels)

Recommended marketing strategies

  1. Premium / high-value cluster - loyalty tiers, early access, premium and investment products, dedicated account managers.
  2. Mid-career (36-55) married professionals - family bundles, long-term plans, cross-sell insurance / savings / home products.
  3. Young customers (17-25), low income - student discounts, app and social-media campaigns, low-cost entry products to build loyalty early.
  4. Part-time / low-hours segment - flexible payments, EMI options and budget products.
  5. High education but not yet high income - career-growth and upskilling offers; they are the likeliest to move into the premium segment.
  6. Other-country / unknown-location customers - localised, multi-language campaigns and region-specific offers.
